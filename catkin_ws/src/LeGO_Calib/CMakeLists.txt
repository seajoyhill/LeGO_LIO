cmake_minimum_required(VERSION 3.0.2)
project(lego_calib)
set(CMAKE_BUILD_TYPE Release)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-O3 -Wall -Wextra -Wpedantic)
add_definitions(-DROOT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/")
find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs nav_msgs geometry_msgs pcl_ros pcl_conversions eigen_conversions tf roslib)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common filters segmentation)
find_package(Ceres REQUIRED)
catkin_package(INCLUDE_DIRS include CATKIN_DEPENDS roscpp sensor_msgs nav_msgs geometry_msgs pcl_ros pcl_conversions eigen_conversions tf roslib DEPENDS EIGEN3 PCL Ceres)
include_directories(include include/Fusion ${catkin_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS})
add_definitions(${PCL_DEFINITIONS})
add_executable(lego_calib_node src/lego_calib_node.cpp src/Gril_Calib.cpp
  src/FusionAhrs.c src/FusionCompass.c src/FusionOffset.c)
add_dependencies(lego_calib_node ${catkin_EXPORTED_TARGETS})
target_link_libraries(lego_calib_node ${catkin_LIBRARIES} ${PCL_LIBRARIES} ${CERES_LIBRARIES})
